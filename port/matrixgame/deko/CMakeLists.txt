cmake_minimum_required(VERSION 3.20)
project(MatrixGameNativeDeko LANGUAGES CXX)
if(NOT CMAKE_CROSSCOMPILING OR NOT MATRIXGAME_DEVKITPRO)
  message(FATAL_ERROR "Native deko3d requires the Switch toolchain")
endif()
if(NOT EXISTS "${MATRIXGAME_SHADER_DIR}/mesh_shaders.hpp")
  message(FATAL_ERROR "Generate shaders with tools/matrixgame_deko_build.py")
endif()
get_filename_component(PORT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../native-core" "${CMAKE_CURRENT_BINARY_DIR}/core")
add_library(matrixgame-native-deko STATIC mesh_renderer.cpp)
target_include_directories(matrixgame-native-deko PUBLIC "${CMAKE_CURRENT_LIST_DIR}"
  "${MATRIXGAME_DEVKITPRO}/libnx/include")
target_link_directories(matrixgame-native-deko PUBLIC "${MATRIXGAME_DEVKITPRO}/libnx/lib")
target_link_libraries(matrixgame-native-deko PUBLIC matrixgame-native-core deko3d nx)
add_executable(matrixgame-deko-mesh-probe "${PORT_ROOT}/tests/probe_matrixgame_deko_link.cpp")
target_include_directories(matrixgame-deko-mesh-probe PRIVATE "${MATRIXGAME_SHADER_DIR}")
target_link_libraries(matrixgame-deko-mesh-probe PRIVATE matrixgame-native-deko)
target_link_options(matrixgame-deko-mesh-probe PRIVATE
  "-specs=${MATRIXGAME_DEVKITPRO}/libnx/switch.specs" -fPIE -Wl,--gc-sections -Wl,--build-id=sha1)
set_target_properties(matrixgame-deko-mesh-probe PROPERTIES SUFFIX ".elf")
