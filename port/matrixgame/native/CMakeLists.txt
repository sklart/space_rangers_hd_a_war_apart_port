  set(SANITIZER_FLAGS -fsanitize=address,undefined -fno-omit-frame-pointer)

  set(SANITIZER_FLAGS -fsanitize=thread -fno-omit-frame-pointer)
elseif(NOT BRIDGE_SANITIZER STREQUAL "none")
  message(FATAL_ERROR "Unknown BRIDGE_SANITIZER")
endif()

option(BRIDGE_BUILD_HOST_LINK_PROBE "Link real Rangers host services and MatrixGame adapter; no NRO" OFF)
set(BRIDGE_RANGERS_HOST_ARCHIVE "" CACHE FILEPATH "Verified original Rangers SDK dependency archive")
set(BRIDGE_RANGERS_OKGF_ARCHIVE "" CACHE FILEPATH "Verified original OKGF SDK archive")
set(BRIDGE_RANGERS_SOFTFLOAT_ARCHIVE "" CACHE FILEPATH "Verified original OKGF softfloat SDK archive")
set(BRIDGE_HOST_LINK_PROBE_SOURCE "${PORT_ROOT}/tests/probe_matrixgame_rangers_host_link.cpp" CACHE FILEPATH "Real host-services diagnostic entry source")
if(BRIDGE_BUILD_HOST_LINK_PROBE)
  if(NOT CMAKE_CROSSCOMPILING OR NOT MATRIXGAME_DEVKITPRO)
    message(FATAL_ERROR "BRIDGE_BUILD_HOST_LINK_PROBE requires Switch toolchain")
  endif()
  if(NOT EXISTS "${BRIDGE_RANGERS_HOST_ARCHIVE}" OR NOT EXISTS "${BRIDGE_RANGERS_OKGF_ARCHIVE}" OR NOT EXISTS "${BRIDGE_RANGERS_SOFTFLOAT_ARCHIVE}")
    message(FATAL_ERROR "Prepare verified dependency archive with tools/matrixgame_rangers_host_sdk.py")
  endif()
  add_executable(matrixgame-rangers-host-link-probe "${BRIDGE_HOST_LINK_PROBE_SOURCE}")
  target_include_directories(matrixgame-rangers-host-link-probe PRIVATE
    "${PORT_ROOT}/upstream/cpp/src" "${PORT_ROOT}/upstream/cpp/runtime" "${PORT_ROOT}/port/switch/platform"
    "${MATRIXGAME_DEVKITPRO}/portlibs/switch/include")
  target_compile_options(matrixgame-rangers-host-link-probe PRIVATE -UNDEBUG)
  target_link_directories(matrixgame-rangers-host-link-probe PRIVATE
    "${MATRIXGAME_DEVKITPRO}/libnx/lib" "${MATRIXGAME_DEVKITPRO}/portlibs/switch/lib")
  target_link_libraries(matrixgame-rangers-host-link-probe PRIVATE
    -Wl,--start-group matrixgame-engine-callbacks matrixgame-rangers-bridge matrixgame-native-core
    "${BRIDGE_RANGERS_HOST_ARCHIVE}" "${BRIDGE_RANGERS_OKGF_ARCHIVE}" "${BRIDGE_RANGERS_SOFTFLOAT_ARCHIVE}"
    png z jpeg vorbisfile vorbis ogg m SDL2 EGL glapi drm_nouveau nx pthread -Wl,--end-group)
  target_link_options(matrixgame-rangers-host-link-probe PRIVATE
    "-specs=${MATRIXGAME_DEVKITPRO}/libnx/switch.specs" -fPIE -Wl,--gc-sections -Wl,--build-id=sha1
    "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/rangers-host-link.map")
  set_target_properties(matrixgame-rangers-host-link-probe PROPERTIES SUFFIX ".elf")
endif()
