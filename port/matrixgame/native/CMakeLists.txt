cmake_minimum_required(VERSION 3.20)
project(MatrixGameRangersBridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
get_filename_component(PORT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_library(matrixgame-rangers-bridge STATIC rangers_bridge.cpp)
target_include_directories(matrixgame-rangers-bridge PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
target_include_directories(matrixgame-rangers-bridge PRIVATE
  "${PORT_ROOT}/upstream/cpp/src" "${PORT_ROOT}/upstream/cpp/runtime")
target_link_libraries(matrixgame-rangers-bridge PUBLIC Threads::Threads)
set(BRIDGE_SANITIZER "none" CACHE STRING "none, address, or thread")
set_property(CACHE BRIDGE_SANITIZER PROPERTY STRINGS none address thread)
if(BRIDGE_SANITIZER STREQUAL "address")
  set(SANITIZER_FLAGS -fsanitize=address,undefined -fno-omit-frame-pointer)
elseif(BRIDGE_SANITIZER STREQUAL "thread")
  set(SANITIZER_FLAGS -fsanitize=thread -fno-omit-frame-pointer)
elseif(NOT BRIDGE_SANITIZER STREQUAL "none")
  message(FATAL_ERROR "Unknown BRIDGE_SANITIZER")
endif()
target_compile_options(matrixgame-rangers-bridge PRIVATE ${SANITIZER_FLAGS})
target_link_options(matrixgame-rangers-bridge PUBLIC ${SANITIZER_FLAGS})
include(CTest)
if(BUILD_TESTING)
  add_executable(matrixgame-rangers-bridge-test "${PORT_ROOT}/tests/test_matrixgame_rangers_bridge.cpp")
  target_include_directories(matrixgame-rangers-bridge-test PRIVATE
    "${PORT_ROOT}/upstream/cpp/src" "${PORT_ROOT}/upstream/cpp/runtime")
  target_compile_options(matrixgame-rangers-bridge-test PRIVATE ${SANITIZER_FLAGS} -UNDEBUG)
  target_link_libraries(matrixgame-rangers-bridge-test PRIVATE matrixgame-rangers-bridge)
  add_test(NAME matrixgame-rangers-bridge-contract COMMAND matrixgame-rangers-bridge-test)
  add_executable(matrixgame-mesh-submission-test "${PORT_ROOT}/tests/test_matrixgame_mesh_submission.cpp")
  target_compile_options(matrixgame-mesh-submission-test PRIVATE ${SANITIZER_FLAGS} -UNDEBUG)
  target_link_options(matrixgame-mesh-submission-test PRIVATE ${SANITIZER_FLAGS})
  add_test(NAME matrixgame-mesh-submission-contract COMMAND matrixgame-mesh-submission-test)
  add_executable(matrixgame-texture-upload-test "${PORT_ROOT}/tests/test_matrixgame_texture_upload.cpp")
  target_compile_options(matrixgame-texture-upload-test PRIVATE ${SANITIZER_FLAGS} -UNDEBUG)
  target_link_options(matrixgame-texture-upload-test PRIVATE ${SANITIZER_FLAGS})
  add_test(NAME matrixgame-texture-upload-contract COMMAND matrixgame-texture-upload-test)
endif()
