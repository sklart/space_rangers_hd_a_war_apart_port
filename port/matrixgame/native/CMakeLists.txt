cmake_minimum_required(VERSION 3.20)
project(MatrixGameRangersBridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_CROSSCOMPILING)
  find_package(Threads REQUIRED)
endif()
get_filename_component(PORT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_library(matrixgame-rangers-bridge STATIC rangers_bridge.cpp)
target_include_directories(matrixgame-rangers-bridge PUBLIC "${CMAKE_CURRENT_LIST_DIR}")
target_include_directories(matrixgame-rangers-bridge PRIVATE
  "${PORT_ROOT}/upstream/cpp/src" "${PORT_ROOT}/upstream/cpp/runtime")
if(TARGET Threads::Threads)
  target_link_libraries(matrixgame-rangers-bridge PUBLIC Threads::Threads)
endif()
if(MSVC)
  target_compile_options(matrixgame-rangers-bridge PUBLIC /utf-8)
endif()
set(BRIDGE_SANITIZER "none" CACHE STRING "none, address, or thread")
set_property(CACHE BRIDGE_SANITIZER PROPERTY STRINGS none address thread)
if(BRIDGE_SANITIZER STREQUAL "address")
  set(SANITIZER_FLAGS -fsanitize=address,undefined -fno-omit-frame-pointer)
elseif(BRIDGE_SANITIZER STREQUAL "thread")
  set(SANITIZER_FLAGS -fsanitize=thread -fno-omit-frame-pointer)
elseif(NOT BRIDGE_SANITIZER STREQUAL "none")
  message(FATAL_ERROR "Unknown BRIDGE_SANITIZER")
endif()
target_compile_options(matrixgame-rangers-bridge PRIVATE ${SANITIZER_FLAGS})
target_link_options(matrixgame-rangers-bridge PUBLIC ${SANITIZER_FLAGS})
if(BRIDGE_SANITIZER STREQUAL "address")
  set(CORE_SANITIZERS ON CACHE BOOL "Instrument CPU dependencies with ASan/UBSan")
elseif(BRIDGE_SANITIZER STREQUAL "thread")
  set(CORE_THREAD_SANITIZER ON CACHE BOOL "Instrument CPU dependencies with TSan")
endif()
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../native-core" "${CMAKE_CURRENT_BINARY_DIR}/core")
if(CMAKE_CROSSCOMPILING)
  # newlib strict C++ mode hides the real POSIX declarations used by pas/text.hpp.
  target_compile_definitions(matrixgame-rangers-bridge PUBLIC _POSIX_C_SOURCE=200809L)
endif()
add_library(matrixgame-engine-callbacks STATIC engine_callbacks.cpp)
target_link_libraries(matrixgame-engine-callbacks PUBLIC matrixgame-rangers-bridge matrixgame-native-core)
target_compile_options(matrixgame-engine-callbacks PRIVATE ${SANITIZER_FLAGS})
include(CTest)
if(BUILD_TESTING)
  add_executable(matrixgame-rangers-bridge-test "${PORT_ROOT}/tests/test_matrixgame_rangers_bridge.cpp")
  target_include_directories(matrixgame-rangers-bridge-test PRIVATE
    "${PORT_ROOT}/upstream/cpp/src" "${PORT_ROOT}/upstream/cpp/runtime")
  target_compile_options(matrixgame-rangers-bridge-test PRIVATE ${SANITIZER_FLAGS} -UNDEBUG)
  target_link_libraries(matrixgame-rangers-bridge-test PRIVATE matrixgame-engine-callbacks)
  add_test(NAME matrixgame-rangers-bridge-contract COMMAND matrixgame-rangers-bridge-test)
  add_executable(matrixgame-mesh-submission-test "${PORT_ROOT}/tests/test_matrixgame_mesh_submission.cpp")
  target_compile_options(matrixgame-mesh-submission-test PRIVATE ${SANITIZER_FLAGS} -UNDEBUG)
  target_link_options(matrixgame-mesh-submission-test PRIVATE ${SANITIZER_FLAGS})
  add_test(NAME matrixgame-mesh-submission-contract COMMAND matrixgame-mesh-submission-test)
  add_executable(matrixgame-texture-upload-test "${PORT_ROOT}/tests/test_matrixgame_texture_upload.cpp")
  target_compile_options(matrixgame-texture-upload-test PRIVATE ${SANITIZER_FLAGS} -UNDEBUG)
  target_link_options(matrixgame-texture-upload-test PRIVATE ${SANITIZER_FLAGS})
  add_test(NAME matrixgame-texture-upload-contract COMMAND matrixgame-texture-upload-test)
endif()

option(BRIDGE_BUILD_LINK_PROBE "SDK native interface link proof; no NRO or battle" OFF)
if(BRIDGE_BUILD_LINK_PROBE)
  if(NOT CMAKE_CROSSCOMPILING OR NOT MATRIXGAME_DEVKITPRO)
    message(FATAL_ERROR "BRIDGE_BUILD_LINK_PROBE requires Switch toolchain")
  endif()
  add_executable(matrixgame-engine-api-link-probe "${PORT_ROOT}/tests/test_matrixgame_rangers_bridge.cpp")
  target_compile_definitions(matrixgame-engine-api-link-probe PRIVATE MATRIXGAME_BRIDGE_NO_THREADS)
  target_include_directories(matrixgame-engine-api-link-probe PRIVATE "${PORT_ROOT}/upstream/cpp/src" "${PORT_ROOT}/upstream/cpp/runtime")
  target_link_libraries(matrixgame-engine-api-link-probe PRIVATE matrixgame-engine-callbacks nx)
  target_link_directories(matrixgame-engine-api-link-probe PRIVATE "${MATRIXGAME_DEVKITPRO}/libnx/lib")
  target_link_options(matrixgame-engine-api-link-probe PRIVATE "-specs=${MATRIXGAME_DEVKITPRO}/libnx/switch.specs" -fPIE -Wl,--gc-sections -Wl,--build-id=sha1)
  target_compile_options(matrixgame-engine-api-link-probe PRIVATE -UNDEBUG)
  set_target_properties(matrixgame-engine-api-link-probe PROPERTIES SUFFIX ".elf")
endif()
